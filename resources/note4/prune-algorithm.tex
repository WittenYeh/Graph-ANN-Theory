\begin{varwidth}{\linewidth}
\setcounter{algocf}{1}
\begin{algorithm}[H]
\SetAlgoLined
\DontPrintSemicolon
\caption{$\textsc{Prune}(p, \mathcal{E}, \alpha, R)$}
\KwIn{Graph $G$, point $p \in P$, candidate set $\mathcal{E} \subseteq P$, distance threshold $\alpha \ge 1$, degree bound $R$}
\KwOut{$G$ modified by setting at most $R$ new out-neighbors for $p$}
$\mathcal{E} \leftarrow \big(\mathcal{E} \cup N_{\mathrm{out}}(p)\big) \setminus \{p\}$\;
$N_{\mathrm{out}}(p) \leftarrow \emptyset$\;
\While{$\mathcal{E} \neq \emptyset$}{
  $p^{*} \leftarrow \arg\min_{p' \in \mathcal{E}} D(p, p')$\;
  $N_{\mathrm{out}}(p) \leftarrow N_{\mathrm{out}}(p) \cup \{p^{*}\}$\;
  \If{$|N_{\mathrm{out}}(p)| = R$}{
    \textbf{break}\;
  }
  \For{$p' \in \mathcal{E}$}{
    \If{$\alpha \cdot D(p^{*}, p') \le D(p, p')$}{
      $\mathcal{E} \leftarrow \mathcal{E} \setminus \{p'\}$\;
    }
  }
}
\end{algorithm}
\end{varwidth}
